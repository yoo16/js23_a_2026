    解説
    
      
      
        
    • 一覧: GET/api/feeds。JOIN で投稿者名を結合し、ORDER BY created_at DESC で新しい順に all() で取得します。
      • 
        
    • Post: POST/api/feeds。INSERT ... VALUES (?, ?) を run() で実行し、戻り値の lastInsertRowid が追加した行の ID です。
      • 
        
    • 存在しない user_id を指定すると、PRAGMA foreign_keys = ON により FOREIGN KEY constraint failed のエラーになります。
      • 
        
    • Edit: POST/api/feeds/:id/update。UPDATE feeds SET content = ? WHERE id = ? を run() で実行します。WHERE を付け忘れると全件が更新されます。
      • 
        
    • Delete: POST/api/feeds/:id/destroy。DELETE ... WHERE id = ? を run() で実行し、削除件数が changes に入ります。
      • 
        
    • COMMIT / ROLLBACK: 2件の INSERT を BEGIN ～ COMMIT でまとめます。ROLLBACK のボタンは2件目の前にわざと失敗させ、1件目も取り消されることを確認できます。
      • 
        
    • Post injection: 入力を文字列連結して exec() に渡すデモです。exec() は複数の SQL を実行できるため、feeds が全件削除されます。値は必ず ? で渡しましょう。

    // models/Feed.js: 一覧 (JOIN + 新しい順)
export const fetchAll = (limit = 20) => {
    const sql = `SELECT feeds.id, feeds.user_id, feeds.content, feeds.created_at,
       users.name AS user_name
FROM feeds
JOIN users ON feeds.user_id = users.id
ORDER BY feeds.created_at DESC, feeds.id DESC
LIMIT ?;`;
    const statement = db.prepare(sql);
    const feeds = statement.all(limit);
    return { feeds, sql: format(sql, [limit]) };
};

// models/Feed.js: 追加
export const insert = ({ user_id, content }) => {
    const sql = "INSERT INTO feeds (user_id, content) VALUES (?, ?);";
    try {
        const statement = db.prepare(sql);
        const result = statement.run(Number(user_id), content);
        const id = Number(result.lastInsertRowid);
        return { id, errors: [] };
    } catch (error) {
        return { errors: [{ msg: error.message }] };   // 外部キー違反など
    }
};

// models/Feed.js: 更新 (WHERE を忘れない)
export const update = (id, content) => {
    const sql = "UPDATE feeds SET content = ? WHERE id = ?;";
    const statement = db.prepare(sql);
    const result = statement.run(content, Number(id));
    const errors = result.changes ? [] : [{ msg: 'Feed not found' }];
    return { id: Number(id), changes: result.changes, errors };
};

// models/Feed.js: 削除
export const destroy = (id) => {
    const sql = "DELETE FROM feeds WHERE id = ?;";
    const statement = db.prepare(sql);
    const result = statement.run(Number(id));
    return { id, changes: result.changes, errors: [] };
};

// models/Feed.js: トランザクション (BEGIN → 複数の SQL → COMMIT / 失敗時は ROLLBACK)
export const insertTwice = ({ user_id, fail = false }) => {
    const sql = "INSERT INTO feeds (user_id, content) VALUES (?, ?);";
    const statement = db.prepare(sql);
    db.exec('BEGIN');
    try {
        statement.run(Number(user_id), 'トランザクション1');
        if (fail) throw new Error('わざと失敗させました');
        statement.run(Number(user_id), 'トランザクション2');
        db.exec('COMMIT');
        return { errors: [] };
    } catch (error) {
        db.exec('ROLLBACK');   // 1件目の INSERT も取り消される
        return { errors: [{ msg: error.message }] };
    }
};
